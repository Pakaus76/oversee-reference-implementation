  304: def write_layer_artifact_json(path: Path, payload: Any) -> None:
  305:     """Write one JSON layer artifact to disk.
  306:     
  307:     This helper persists already-computed Python objects as JSON files for review,
  308:     traceability and auditability. It does not contain OVERSEE decision logic; it
  309:     only serializes the artifacts produced by the runner.
  310:     """

  312:     path.write_text(
  313:         json.dumps(
  314:             payload,
  315:             indent=2,
  316:             ensure_ascii=False,
  317:             default=serialize_layer_artifact_default,
  318:         ),
  319:         encoding="utf-8",
  320:     )

  323: def serialize_layer_artifact_default(value: Any) -> Any:
  324:     """Convert common OVERSEE layer objects into JSON-serializable values.
  325:     
  326:     The all-layers runner receives dataclasses, objects with to_dict() methods and
  327:     standard Python values. This serializer keeps artifact writing centralized and
  328:     consistent across layers.
  329:     """
  330: 
  331:     if hasattr(value, "to_dict"):
  332:         return value.to_dict()
  333: 
  334:     if is_dataclass(value):
  335:         return asdict(value)
  336: 
  337:     raise TypeError(f"Object of type {value.__class__.__name__} is not JSON serializable")

  210: def _action_alignment(
  211:     deterministic_action: str | None,
  212:     generative_action: str | None,
  213: ) -> str:
  214:     """Classify action alignment with a simple token-overlap heuristic."""
  215: 
  216:     if not deterministic_action or not generative_action:
  217:         return "missing"
  218: 
  219:     det_tokens = _content_tokens(deterministic_action)
  220:     gen_tokens = _content_tokens(generative_action)
  221: 
  222:     if not det_tokens or not gen_tokens:
  223:         return "missing"
  224: 
  225:     overlap = len(det_tokens.intersection(gen_tokens))
  226:     ratio = overlap / max(len(det_tokens), len(gen_tokens))
  227: 
  228:     if ratio >= 0.7:
  229:         return "strong"
  230:     if ratio >= 0.35:
  231:         return "partial"
  232:     return "different"

  249:         "with",

  252:     }

  255:         token
