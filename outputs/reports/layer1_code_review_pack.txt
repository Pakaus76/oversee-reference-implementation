  305:     """Write one JSON layer artifact to disk.
  306:     
  307:     This helper persists already-computed Python objects as JSON files for review,

  323: def serialize_layer_artifact_default(value: Any) -> Any:
  324:     """Convert common OVERSEE layer objects into JSON-serializable values.
  325:     
  326:     The all-layers runner receives dataclasses, objects with to_dict() methods and
  327:     standard Python values. This serializer keeps artifact writing centralized and
  328:     consistent across layers.
  329:     """
  330: 
  331:     if hasattr(value, "to_dict"):
  332:         return value.to_dict()
  333: 
  334:     if is_dataclass(value):
  335:         return asdict(value)
  336: 
  337:     raise TypeError(f"Object of type {value.__class__.__name__} is not JSON serializable")
